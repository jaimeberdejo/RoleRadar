    """Tests for app/scoring/puesto_match.py — SCORE-11."""
    from __future__ import annotations
    import numpy as np
    import pytest
    from app.models.schemas import PuestoRanking
    from app.scoring.puesto_match import match_puesto_por_coseno, UMBRAL_PUESTO_DEFAULT


    def _ranking() -> list[PuestoRanking]:
        return [
            PuestoRanking(titulo="AI Engineer", sinonimos=["LLM Engineer"]),
            PuestoRanking(titulo="Data Engineer", sinonimos=["DE"]),
        ]


    def test_match_rango_1_identical_vectors() -> None:
        """Identical vectors → best match is rango=1 (first entry)."""
        from app.dedup.embedder import FakeEmbedder
        vec = np.array([1.0, 0.0, 0.0, 0.0], dtype=np.float32)
        embedder = FakeEmbedder(default_vector=vec)
        puesto, rango = match_puesto_por_coseno("AI Engineer", _ranking(), embedder)
        assert rango == 1
        assert puesto == "AI Engineer"

    def test_match_rango_2() -> None:
        """Second ranking entry is closer → rango=2."""
        from app.dedup.embedder import FakeEmbedder
        # job title vector similar to Data Engineer (second entry)
        vecs = {
            "Data Engineer": np.array([0.0, 1.0, 0.0, 0.0], dtype=np.float32),
            "AI Engineer LLM Engineer": np.array([1.0, 0.0, 0.0, 0.0], dtype=np.float32),
            "Data Engineer DE": np.array([0.0, 1.0, 0.0, 0.0], dtype=np.float32),
        }
        embedder = FakeEmbedder(vectors=vecs, default_vector=np.array([0.0, 1.0, 0.0, 0.0], dtype=np.float32))
        # Use the title that maps to second entry's vector
        puesto, rango = match_puesto_por_coseno("Data Engineer", _ranking(), embedder)
        assert rango == 2
        assert puesto == "Data Engineer"

    def test_below_threshold_fuera_de_ranking() -> None:
        """Below threshold → fuera de ranking, rango=None."""
        from app.dedup.embedder import FakeEmbedder
        vec_title = np.array([1.0, 0.0, 0.0, 0.0], dtype=np.float32)
        vec_ranking = np.array([0.0, 1.0, 0.0, 0.0], dtype=np.float32)  # orthogonal
        vecs = {
            "Something Else": vec_title,
            "AI Engineer LLM Engineer": vec_ranking,
            "Data Engineer DE": vec_ranking,
        }
        embedder = FakeEmbedder(vectors=vecs)
        puesto, rango = match_puesto_por_coseno("Something Else", _ranking(), embedder, umbral=0.99)
        assert rango is None
        assert puesto == "fuera de ranking"

    def test_embedder_none_raises() -> None:
        """embedder=None raises ValueError with 'embedder' in message."""
        with pytest.raises(ValueError, match="embedder"):
            match_puesto_por_coseno("AI Engineer", _ranking(), None)

    def test_empty_ranking() -> None:
        """Empty ranking → fuera de ranking, rango=None."""
        from app.dedup.embedder import FakeEmbedder
        embedder = FakeEmbedder(default_vector=np.array([1.0, 0.0, 0.0, 0.0], dtype=np.float32))
        puesto, rango = match_puesto_por_coseno("AI Engineer", [], embedder)
        assert rango is None
        assert puesto == "fuera de ranking"

    def test_umbral_default_exported() -> None:
        assert UMBRAL_PUESTO_DEFAULT == 0.55
